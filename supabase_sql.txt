-- Create Feedback table
CREATE TABLE IF NOT EXISTS public.feedback (
    id bigint generated by default as identity primary key,
    created_at timestamp with time zone default timezone('utc'::text, now()) not null,
    customer_name text,
    rating integer,
    content text
);

ALTER TABLE public.feedback ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Enable insert for anon" ON public.feedback FOR INSERT WITH CHECK (true);
CREATE POLICY "Enable read for authenticated" ON public.feedback FOR SELECT USING (true);

-- Create Kashrut images table
CREATE TABLE IF NOT EXISTS public.kashrut_images (
    id bigint generated by default as identity primary key,
    created_at timestamp with time zone default timezone('utc'::text, now()) not null,
    image_url text not null,
    title text
);

ALTER TABLE public.kashrut_images ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Enable read for everyone" ON public.kashrut_images FOR SELECT USING (true);
CREATE POLICY "Enable insert/update for authenticated" ON public.kashrut_images FOR ALL USING (true);

-- Create images bucket and policies if not exist
INSERT INTO storage.buckets (id, name, public) VALUES ('images', 'images', true) ON CONFLICT (id) DO NOTHING;

CREATE POLICY "Public Access" ON storage.objects FOR SELECT USING (bucket_id = 'images');
CREATE POLICY "Anon Insert" ON storage.objects FOR INSERT WITH CHECK (bucket_id = 'images');
